cmake_minimum_required(VERSION 3.18)
project(aikeyboard CXX C)
set(CMAKE_CXX_STANDARD 17)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_COMMON OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(LLAMA_CURL OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_LLAMAFILE ON CACHE BOOL "" FORCE)
add_subdirectory(../src/llama.cpp llama EXCLUDE_FROM_ALL)
if(ANDROID)
  add_library(aikeyboard SHARED aicore.cpp jni.cpp)
  target_link_libraries(aikeyboard llama ggml log android)
  target_link_options(aikeyboard PRIVATE -Wl,-z,max-page-size=16384 -Wl,--gc-sections -Wl,--exclude-libs,ALL)
endif()
add_executable(aik_cli_any aicore.cpp cli.cpp)
target_link_libraries(aik_cli_any llama ggml)
if(NOT ANDROID AND DEFINED JNI_INC)
  add_library(aikeyboard SHARED aicore.cpp jni.cpp)
  target_include_directories(aikeyboard PRIVATE ${JNI_INC} ${JNI_INC}/linux)
  target_link_libraries(aikeyboard llama ggml)
endif()
