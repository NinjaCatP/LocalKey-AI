cmake_minimum_required(VERSION 3.18)
project(aikvoice CXX C)
set(CMAKE_CXX_STANDARD 17)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(WHISPER_SDL2 OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
add_subdirectory(../../src/whisper.cpp whisper EXCLUDE_FROM_ALL)
add_library(aikvoice SHARED voice.cpp)
if(ANDROID)
  target_link_libraries(aikvoice whisper ggml log)
  target_link_options(aikvoice PRIVATE -Wl,-z,max-page-size=16384 -Wl,--gc-sections -Wl,--exclude-libs,ALL)
else()
  target_include_directories(aikvoice PRIVATE ${JNI_INC} ${JNI_INC}/linux)
  target_link_libraries(aikvoice whisper ggml)
  target_link_options(aikvoice PRIVATE -Wl,--exclude-libs,ALL)
endif()
add_executable(aikv_cli voice.cpp cli.cpp)
target_include_directories(aikv_cli PRIVATE ${JNI_INC} ${JNI_INC}/linux)
target_link_libraries(aikv_cli whisper ggml)
